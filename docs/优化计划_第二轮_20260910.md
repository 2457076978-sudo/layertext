### A.1 加注覆盖率审计（本轮 H1 的测量工具）

放到 `tools/` 下即可复用；它对每一章每一层算出"应注 OOV 词型 / 实际已注"。

```js
// tools/annotation_coverage.mjs
import { readFileSync } from 'node:fs';
const LTR = process.env.LAYERTEXT_ENGINE ?? '/绝对路径/LayerText';
const KF  = '/绝对路径/…/知识文件';
const OUT = '/绝对路径/…/调适工作区/重制三版';

const { runQc } = await import(`${LTR}/dist/src/core/qc.js`);
const { buildLexicon } = await import(`${LTR}/dist/src/core/lexicon.js`);
const LEX = buildLexicon({ vocabCsvTexts: [readFileSync(`${KF}/已知词汇库_v0.7.csv`, 'utf-8')] });
const PROPER = readFileSync(`${KF}/专名_AnimalFarm.txt`, 'utf-8')
  .split('\n').map((s) => s.trim()).filter((s) => s && !s.startsWith('#'));

const CH = ['第一章','第二章','第三章','第四章','第五章','第六章','第七章','第八章','第九章','第十章'];
const TIERS = { A: 'A层85', M: 'M层75', B: 'B层60' };
let bad = 0;

for (const [k, tag] of Object.entries(TIERS)) {
  let need = 0, done = 0;
  for (const ch of CH) {
    const md = readFileSync(`${OUT}/${ch}/原文_${tag}_2026-09-10.md`, 'utf-8');
    const qc = runQc(md, LEX, { tier: k, fileName: 'x.md', properNouns: PROPER });
    const oov = [...new Set(qc.oov)].filter((w) => w.length > 2 && !PROPER.includes(w));
    const miss = oov.filter((w) => !md.includes(`${w}（`));
    need += oov.length; done += oov.length - miss.length;
    const rate = oov.length ? (oov.length - miss.length) / oov.length : 1;
    if (rate < 0.7) { bad++; console.log(`❌ ${ch} ${tag} 覆盖率 ${(rate*100).toFixed(0)}%（缺 ${miss.length}）`); }
  }
  console.log(`${tag}：${done}/${need} = ${(done/need*100).toFixed(0)}%`);
}
process.exit(bad ? 1 : 0);   // 可直接当 CI 闸门
```

**本次实测结果**（就是 H1 的证据）：

```
A层85：800/1800 = 44%   ← 第七章 5/255、第八章 11/289、第九章 3/250
M层75：923/1262 = 73%
B层60：605/884  = 68%
```
